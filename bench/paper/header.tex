% Wide result tables are the reason for most of this preamble.
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{ragged2e}
\usepackage{float}
\usepackage{caption}
\captionsetup{font=small,labelfont=bf}
% Caption text already starts with its own bold "Figure N." / "Table N."
% (prepare_body.py moves it into the float), so LaTeX's label would repeat it.
\captionsetup[figure]{labelformat=empty}
\captionsetup[table]{labelformat=empty}

% Tables in this paper are numerous and wide; shrink them rather than let them
% run into the margin.
\AtBeginEnvironment{longtable}{\small}
\AtBeginEnvironment{tabular}{\small}

% Figures float. Pinning them in place (H) was how captions used to stay beside
% them; captions are now inside the float (prepare_body.py), and H left the page
% before a figure that did not fit stretched to fill it under \flushbottom.
\floatplacement{figure}{htbp}

% Code spans appear constantly (flag names, file paths). Make them tight.
\let\oldtexttt\texttt
\renewcommand{\texttt}[1]{{\small\oldtexttt{#1}}}

\usepackage{microtype}
\setlength{\emergencystretch}{3em}
